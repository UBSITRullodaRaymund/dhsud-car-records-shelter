function listUsers(requestingUser) {
  const sheet = getOrCreateUsersSheet();
  const data = sheet.getDataRange().getValues();

  if (data.length < 2) return jsonResponse([]);

  const headers = data[0];
  let users = data.slice(1).map(row =>
    Object.fromEntries(headers.map((header, i) => [header, row[i]]))
  );

  users = users.map(user => ({
    email: user.Email,
    fullName: user.FullName,
    role: user.Role,
    office: user.Office,
    status: user.Status,
    permissions: parsePermissions(user.Permissions)
  }));

  if (requestingUser.role !== "SuperAdmin") {
    users = users.filter(user => user.office === requestingUser.office);
  }

  return jsonResponse(users);
}

function createUserAccount(e, requestingUser) {
  const email = String(e.parameter.email || "").trim().toLowerCase();
  const password = String(e.parameter.password || "");
  const fullName = String(e.parameter.fullName || "");
  const role = String(e.parameter.role || "OfficeStaff");
  const office = String(e.parameter.office || "");
  const status = String(e.parameter.status || "Active");
  const permissions = String(e.parameter.permissions || "");

  if (!email || !password || !office) {
    return jsonResponse({
      error: "Email, password, and office are required"
    }, 400);
  }

  if (
    requestingUser.role !== "SuperAdmin" &&
    office !== requestingUser.office
  ) {
    return jsonResponse({
      error: "You can only create accounts for your own office"
    }, 403);
  }

  const sheet = getOrCreateUsersSheet();
  const data = sheet.getDataRange().getValues();
  const headers = data[0];
  const emailColumn = headers.indexOf("Email");

  const exists = data.some((row, index) =>
    index > 0 &&
    String(row[emailColumn]).toLowerCase() === email
  );

  if (exists) {
    return jsonResponse({
      error: "An account with this email already exists"
    }, 409);
  }

  const salt = generateSalt();
  const passwordHash = hashPassword(password, salt);

  sheet.appendRow([
    email,
    passwordHash,
    salt,
    fullName,
    role,
    office,
    status,
    new Date(),
    permissions
  ]);

  return jsonResponse({ success: true });
}

function updateUserAccount(e, requestingUser) {
  const email = String(e.parameter.email || "").trim().toLowerCase();

  if (!email) {
    return jsonResponse({ error: "Email is required" }, 400);
  }

  const sheet = getOrCreateUsersSheet();
  const data = sheet.getDataRange().getValues();
  const headers = data[0];

  const index = {
    email: headers.indexOf("Email"),
    passwordHash: headers.indexOf("PasswordHash"),
    salt: headers.indexOf("Salt"),
    fullName: headers.indexOf("FullName"),
    role: headers.indexOf("Role"),
    office: headers.indexOf("Office"),
    status: headers.indexOf("Status"),
    permissions: headers.indexOf("Permissions")
  };

  const rowIndex = data.findIndex((row, i) =>
    i > 0 &&
    String(row[index.email]).toLowerCase() === email
  );

  if (rowIndex === -1) {
    return jsonResponse({ error: "Account not found" }, 404);
  }

  const currentOffice = data[rowIndex][index.office];

  if (
    requestingUser.role !== "SuperAdmin" &&
    currentOffice !== requestingUser.office
  ) {
    return jsonResponse({
      error: "You can only manage accounts in your own office"
    }, 403);
  }

  const values = {
    fullName: e.parameter.fullName,
    role: e.parameter.role,
    office: e.parameter.office,
    status: e.parameter.status,
    permissions: e.parameter.permissions
  };

  Object.entries(values).forEach(([field, value]) => {
    if (value !== undefined && index[field] !== -1) {
      sheet.getRange(rowIndex + 1, index[field] + 1).setValue(value);
    }
  });

  if (e.parameter.password) {
    const salt = generateSalt();
    const passwordHash = hashPassword(e.parameter.password, salt);

    sheet.getRange(rowIndex + 1, index.passwordHash + 1)
      .setValue(passwordHash);

    sheet.getRange(rowIndex + 1, index.salt + 1)
      .setValue(salt);
  }

  return jsonResponse({ success: true });
}

function deleteUserAccount(e, requestingUser) {
  const email = String(e.parameter.email || "").trim().toLowerCase();

  if (!email) {
    return jsonResponse({ error: "Email is required" }, 400);
  }

  if (email === requestingUser.email.toLowerCase()) {
    return jsonResponse({
      error: "You cannot delete your own account"
    }, 400);
  }

  const sheet = getOrCreateUsersSheet();
  const data = sheet.getDataRange().getValues();
  const headers = data[0];

  const emailIndex = headers.indexOf("Email");
  const officeIndex = headers.indexOf("Office");

  const rowIndex = data.findIndex((row, i) =>
    i > 0 &&
    String(row[emailIndex]).toLowerCase() === email
  );

  if (rowIndex === -1) {
    return jsonResponse({ error: "Account not found" }, 404);
  }

  if (
    requestingUser.role !== "SuperAdmin" &&
    data[rowIndex][officeIndex] !== requestingUser.office
  ) {
    return jsonResponse({
      error: "You can only manage accounts in your own office"
    }, 403);
  }

  sheet.deleteRow(rowIndex + 1);

  return jsonResponse({ success: true });
}

function handleLogin(e) {
  const email = String(e.parameter.email || "").trim().toLowerCase();
  const password = String(e.parameter.password || "");

  const sheet = getOrCreateUsersSheet();
  const data = sheet.getDataRange().getValues();
  const headers = data.shift();

  const index = {
    email: headers.indexOf("Email"),
    hash: headers.indexOf("PasswordHash"),
    salt: headers.indexOf("Salt"),
    fullName: headers.indexOf("FullName"),
    role: headers.indexOf("Role"),
    office: headers.indexOf("Office"),
    status: headers.indexOf("Status"),
    permissions: headers.indexOf("Permissions")
  };

  const row = data.find(user =>
    String(user[index.email]).toLowerCase() === email
  );

  if (!row) {
    return jsonResponse({
      error: "Invalid email or password"
    }, 401);
  }

  if (String(row[index.status]).trim() !== "Active") {
    return jsonResponse({
      error: "This account is inactive"
    }, 403);
  }

  const computedHash = hashPassword(password, row[index.salt]);

  if (computedHash !== row[index.hash]) {
    return jsonResponse({
      error: "Invalid email or password"
    }, 401);
  }

  const user = {
    email,
    fullName: row[index.fullName],
    role: row[index.role],
    office: row[index.office],
    permissions: parsePermissions(row[index.permissions])
  };

  const token = createSession(user);

  return jsonResponse({
    success: true,
    token,
    user
  });
}

function handleLogout(e) {
  const token = e.parameter.token;

  if (!token) return jsonResponse({ success: true });

  const sheet = getOrCreateSessionsSheet();
  const data = sheet.getDataRange().getValues();
  const headers = data[0];
  const tokenIndex = headers.indexOf("Token");

  for (let i = 1; i < data.length; i++) {
    if (data[i][tokenIndex] === token) {
      sheet.deleteRow(i + 1);
      break;
    }
  }

  return jsonResponse({ success: true });
}

LABEL 9: SESSION FUNCTIONS
============================================================

function createSession(user) {
  const sheet = getOrCreateSessionsSheet();
  const token = Utilities.getUuid();
  const now = new Date();
  const expires = new Date(
    now.getTime() + SESSION_DURATION_MS
  );

  sheet.appendRow([
    token,
    user.email,
    user.role,
    user.office,
    user.fullName,
    now,
    expires,
    (user.permissions || []).join(",")
  ]);

  return token;
}

function requireAuth(token) {
  if (!token) {
    return {
      ok: false,
      error: "Missing session token"
    };
  }

  const sheet = getOrCreateSessionsSheet();
  const data = sheet.getDataRange().getValues();
  const headers = data.shift();
  const tokenIndex = headers.indexOf("Token");
  const row = data.find(session => session[tokenIndex] === token);

  if (!row) {
    return {
      ok: false,
      error: "Invalid session"
    };
  }

  const expiresAt = row[headers.indexOf("ExpiresAt")];
  if (new Date(expiresAt).getTime() < Date.now()) {
    return {
      ok: false,
      error: "Session expired"
    };
  }

  return {
    ok: true,
    user: {
      email: row[headers.indexOf("Email")],
      role: row[headers.indexOf("Role")],
      office: row[headers.indexOf("Office")],
      fullName: row[headers.indexOf("FullName")],
      permissions: parsePermissions(
        row[headers.indexOf("Permissions")]
      )
    }
  };
}

function requireRole(user, allowedRoles) {
  return allowedRoles.includes(user.role);
}

function parsePermissions(value) {
  if (!value) return [];

  return String(value)
    .split(",")
    .map(permission => permission.trim())
    .filter(Boolean);
}

============================================================

function hashPassword(password, salt) {
  const bytes = Utilities.computeDigest(
    Utilities.DigestAlgorithm.SHA_256,
    password + salt,
    Utilities.Charset.UTF_8
  );

  return bytes
    .map(byte => (byte < 0 ? byte + 256 : byte)
      .toString(16)
      .padStart(2, "0"))
    .join("");
}

function generateSalt() {
  return Utilities.getUuid();
}

LABEL 11: USERS AND SESSIONS SHEETS

function getOrCreateUsersSheet() {
  let sheet = SS.getSheetByName(USERS_SHEET_NAME);

  if (!sheet) {
    sheet = SS.insertSheet(USERS_SHEET_NAME);
  }

  ensureSheetHeaders(sheet, [
    "Email",
    "PasswordHash",
    "Salt",
    "FullName",
    "Role",
    "Office",
    "Status",
    "DateCreated",
    "Permissions"
  ]);

  return sheet;
}

function getOrCreateSessionsSheet() {
  let sheet = SS.getSheetByName(SESSIONS_SHEET_NAME);

  if (!sheet) {
    sheet = SS.insertSheet(SESSIONS_SHEET_NAME);
    sheet.appendRow([
      "Token",
      "Email",
      "Role",
      "Office",
      "FullName",
      "CreatedAt",
      "ExpiresAt",
      "Permissions"
    ]);
  }

  return sheet;
}

function ensureSheetHeaders(sheet, requiredHeaders) {
  const values = sheet.getDataRange().getValues();

  if (!values.length) {
    sheet.appendRow(requiredHeaders);
    return;
  }

  const existingHeaders = values[0].map(header =>
    String(header || "").trim()
  );

  const missingHeaders = requiredHeaders.filter(header =>
    !existingHeaders.includes(header)
  );

  if (missingHeaders.length) {
    sheet.getRange(
      1,
      1,
      1,
      existingHeaders.length + missingHeaders.length
    ).setValues([
      existingHeaders.concat(missingHeaders)
    ]);
  }
}
